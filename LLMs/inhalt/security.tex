\textbf{Attacking text-only LLMs}: challenge: input space is discrete; simple pert. set: adv. example $\mathbf{x}' = \mathbf{x} \text{(orig. input)} + \delta \text{(adv. suffix)}$; yields \textbf{Loss function} $\mathcal{L}_{\text{prefix}}(\mathbf{x}+\delta') = \mathcal{L}(y_1, \text{``Sure''}) + \dots + \mathcal{L}(y_n, \text{``that''})$, $\mathcal{L}_{\text{prefix}}(\mathbf{x}+\delta') \approx \mathcal{L}_{\text{prefix}}(\mathbf{x}+\delta) + \nabla_{\mathbf{s}_i}\mathcal{L}'(s_1, \dots, s_n) \cdot (s_i - s_i')$, where $s_i'$ embedding of $i^{\text{th}}$ token in $\delta'$

\textbf{Greedy Coordinate Gradient (GCG) Attack}: computes $\nabla_{\mathbf{s}_i}\mathcal{L}'(s_1, \dots, s_n)$ and picks k tokens that maximize loss approximation; select $B \leq k·n$ of all the candidate token substitutions across all positions; pick token replacement $\delta^{(i)}$ which maximizes $\mathcal{L}_{\text{prefix}}(\mathbf{x}+\delta^{(i)}) $

\subsection{Watermarking}
bias LLM away from true $p(s_{i+1}|s_i)$\\
\textbf{Hard Red-List WM} Randomly partition vocab $V$ into $G$ \& $R$ list for each generated token based on prev. token as seed; model restricted to select tokens from $G$; z-test to eval. whether text has WM: $z=\frac{2(|s|_G-T/2)}{\sqrt{T}}, T=\text{\#tokens}, |s|_G=\text{\#green list tokens used}$; large pos. z $\to$ likely WM

\textbf{Soft Red-List WM} $p_i = \begin{cases}
    \frac{e^{k_i-\delta}}{\sum_{j\in R}e^{k_j-\delta}+\sum_{j\in G}e^{k_j}} & \text{if} \ i \in R\\
    \frac{e^{k_i}}{\sum_{j\in R}e^{k_j-\delta}+\sum_{j\in G}e^{k_j}} & \text{if} \ i \in G
\end{cases} \ , k: \text{logits}$
% subtract constant $\delta$ from logits of red-listed tokens; only enforces WM when multiple good choices available

\textbf{Attacking Transformers}: A: recover hidden dim; 1. query model on $n$ random propmts $p_i$ to obtain logit vets $W\cdot g(p_i)$; 2. construct $Q = WH$ with each column being logit vec; 3. return rank of Q; B: recover embed proj. matrix: 1. SVD of $Q=U\Sigma V^T$; 2. $U\Sigma=WG$ for some G
