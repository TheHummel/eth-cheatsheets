\newif\ifshowmarkers

\global\mdfdefinestyle{added}{%
    topline=false, bottomline=false, rightline=false,
    linecolor=myadded, linewidth=1.5pt,
    backgroundcolor=myadded!6,
    innerleftmargin=4pt, innerrightmargin=2pt,
    innertopmargin=2pt, innerbottommargin=2pt,
    leftmargin=0mm, rightmargin=0mm,
    skipabove=0.5ex, skipbelow=0.5ex,

% block of content added beyond the slides
\newenvironment{added}{%
  \ifshowmarkers
    \begin{mdframed}[style=added]%
    \textsf{\scriptsize\color{myadded}\textbf{[not in slides]}}\par
  \fi
}{%
  \ifshowmarkers\end{mdframed}\fi
}

% short inline addition beyond the slides
\newcommand{\addedin}[1]{%
  \ifshowmarkers{\color{myadded}[+\,}#1{\color{myadded}\,]}\else#1\fi
}

% open question / something unclear in the slides
\newcommand{\unclear}[1]{%
  \ifshowmarkers
    \par\noindent{\sffamily\scriptsize\color{myred}\textbf{[?]}~#1}\par
  \fi
}
